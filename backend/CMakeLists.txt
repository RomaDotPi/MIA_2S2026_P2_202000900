cmake_minimum_required(VERSION 3.16)
project(ExtreamFS LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(extreamfs_core src/extreamfs.cpp)
target_include_directories(extreamfs_core PUBLIC include)
add_executable(extreamfs src/main.cpp)
target_link_libraries(extreamfs PRIVATE extreamfs_core)
if(UNIX)
  target_compile_options(extreamfs_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(extreamfs PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(extreamfs_tests tests/ext3_tests.cpp)
  target_link_libraries(extreamfs_tests PRIVATE extreamfs_core)
  if(UNIX)
    target_compile_options(extreamfs_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME extreamfs_ext2_ext3 COMMAND extreamfs_tests)
  add_executable(extreamfs_official_tests tests/official_evaluation_tests.cpp)
  target_link_libraries(extreamfs_official_tests PRIVATE extreamfs_core)
  if(UNIX)
    target_compile_options(extreamfs_official_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME extreamfs_official_evaluation COMMAND extreamfs_official_tests)
endif()
